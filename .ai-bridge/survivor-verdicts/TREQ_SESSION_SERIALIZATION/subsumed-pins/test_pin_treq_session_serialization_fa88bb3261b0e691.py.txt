from __future__ import annotations

import json

import pytest

from llm_router._internal.session.serialization import decode_session

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.verifies("TREQ_SESSION_SERIALIZATION[revision==1]")
def test_decode_preserves_string_system_prompt() -> None:
    text = json.dumps(
        {"version": 1, "system": "You are a helpful router.", "history": []}
    )

    system, history = decode_session(text)

    assert system == "You are a helpful router."
    assert history == ()
