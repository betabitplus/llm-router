from __future__ import annotations

import json

import pytest

from llm_router import LLMRouter, Model, Provider, RouterProfile
from tests.llm_router.support.fault_server import (
    ScriptedHTTPServer,
    ScriptedResponse,
)
from tests.llm_router.support.workers.retry import (
    openai_chat_path,
    openai_success_response,
)
from tests.llm_router.support.workers.worker_patches import patched_openai_sdk

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.verifies("REQ_REQUEST_OVERRIDE_PRECEDENCE[revision==1]")
def test_kwargs_precedence_route_router_call(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    route_ident = 1
    monkeypatch.setenv(
        f"OPENROUTER_API_KEY_{route_ident}",
        f"value-{route_ident}",
    )
    profile = RouterProfile(
        model=Model.DEEPSEEK_V3,
        provider=Provider.OPENROUTER,
        key_id=route_ident,
        max_attempts=1,
        kwargs={"user": "route-user", "stop": ["route-stop"]},
    )
    router = LLMRouter(
        profile,
        max_attempts=1,
        user="router-user",
        stop=["router-stop"],
        presence_penalty=1.0,
        frequency_penalty=1.0,
        metadata={"tier": "router"},
        shuffle_fallbacks=False,
        round_robin_start=False,
    )
    path = openai_chat_path()
    response = ScriptedResponse(
        status_code=200,
        headers={"Content-Type": "application/json"},
        body=openai_success_response(text="done"),
    )
    with (
        ScriptedHTTPServer(port=0, routes={("POST", path): [response]}) as server,
        patched_openai_sdk(
            forced_base_url=f"{server.base_url}/v1",
            disable_sdk_retries=True,
        ),
    ):
        router.query(
            "hello",
            user="call-user",
            top_p=0.5,
            frequency_penalty=None,
            metadata={},
        )

        (record,) = server.recorded_requests("POST", path)
        recorded = json.loads(record.body)

    assert recorded.get("presence_penalty") == 1.0
    assert recorded.get("stop") == ["router-stop"]
    assert recorded.get("user") == "call-user"
    assert recorded.get("top_p") == 0.5
    assert recorded.get("frequency_penalty") is None
    assert recorded.get("metadata") == {}
