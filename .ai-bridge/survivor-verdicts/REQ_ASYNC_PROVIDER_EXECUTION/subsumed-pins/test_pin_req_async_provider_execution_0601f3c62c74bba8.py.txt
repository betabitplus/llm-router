from __future__ import annotations

import asyncio

import pytest

from llm_router import LLMRouter, Model, Provider, ProviderLimits, RouterProfile

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.asyncio
@pytest.mark.verifies("REQ_ASYNC_PROVIDER_EXECUTION[revision==1]")
async def test_async_briefly_blocked_only_route_never_returns_none(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    value = "mock-provider-auth-12345"
    monkeypatch.setenv("NVIDIA_API_KEY", value)
    limits = ProviderLimits(
        rps=20.0,
        rpm=1200.0,
        cooldown_seconds=0.05,
        cooldown_after_failures=1,
    )
    router = LLMRouter(
        RouterProfile(model=Model.DEEPSEEK_V4_FLASH, provider=Provider.NVIDIA),
        default_limits=limits,
        wait_for_cooldown_if_all_blocked=True,
        max_attempts=1,
        attempt_timeout_seconds=2.0,
    )

    first = await asyncio.gather(router.aquery("first prompt"), return_exceptions=True)
    second = await asyncio.gather(
        router.aquery("second prompt"), return_exceptions=True
    )

    assert len(first) == 1
    assert len(second) == 1
    assert first[0] is not None
    assert second[0] is not None
