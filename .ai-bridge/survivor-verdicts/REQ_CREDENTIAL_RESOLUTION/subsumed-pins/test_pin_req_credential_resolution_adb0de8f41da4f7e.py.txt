from __future__ import annotations

import pytest

from llm_router import LLMRouter, Model, Provider, RouterProfile
from tests.llm_router.support.fault_server import (
    ScriptedHTTPServer,
    ScriptedResponse,
)
from tests.llm_router.support.workers.retry import (
    qwen_chat_path,
    qwen_success_response,
)
from tests.llm_router.support.workers.worker_patches import (
    install_worker_provider_base_url,
)

pytestmark = pytest.mark.verification_kind("unit")

PROVIDER = Provider.QWENCHAT


@pytest.mark.verifies("REQ_CREDENTIAL_RESOLUTION[revision==1]")
def test_auto_key_without_credentials_resolves_permitted_empty_value(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """Auto selection for a provider allowing no bearer credential still works."""
    prefix = f"{PROVIDER.name}_API_KEY"
    monkeypatch.delenv(prefix, raising=False)
    for number in range(100):
        monkeypatch.delenv(f"{prefix}_{number}", raising=False)
    path = qwen_chat_path()
    with ScriptedHTTPServer(
        port=0,
        routes={
            ("POST", path): [
                ScriptedResponse(
                    status_code=200,
                    headers={"Content-Type": "application/json"},
                    body=qwen_success_response(text="ok"),
                )
            ]
        },
    ) as server:
        install_worker_provider_base_url(
            provider="qwenchat", base_url=f"{server.base_url}/api"
        )
        router = LLMRouter(
            RouterProfile(model=Model.QWEN_MAX_LATEST, provider=PROVIDER),
            key_id="auto",
            max_attempts=1,
        )

        response = router.query("hello")

    assert response.output_text == "ok"
    assert response.provider == PROVIDER.value
