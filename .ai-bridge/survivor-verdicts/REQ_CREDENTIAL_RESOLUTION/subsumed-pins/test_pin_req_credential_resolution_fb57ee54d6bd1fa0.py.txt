from __future__ import annotations

import pytest

from llm_router import Provider
from llm_router._internal.config import build_default_config
from llm_router._internal.runtime.limiter import KeyResolver

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.verifies("REQ_CREDENTIAL_RESOLUTION[revision==1]")
def test_auto_rotation_ignores_unrelated_digit_only_env_names(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    config = build_default_config()
    for index in range(100):
        monkeypatch.delenv(f"NVIDIA_API_KEY_{index}", raising=False)
    value = "credential-value"
    monkeypatch.setenv("NVIDIA_API_KEY_1", value)
    monkeypatch.setenv("NVIDIA_API_KEY_2", value)
    monkeypatch.setenv("7", "unrelated")
    resolver = KeyResolver(config)

    candidates = resolver.candidates(provider=Provider.NVIDIA, key_id="auto")
    resolved = [
        resolver.resolve(provider=Provider.NVIDIA, key_id="auto") for _ in range(3)
    ]

    assert [item.key_id for item in candidates] == [1, 2]
    assert [item.key_id for item in resolved] == [1, 2, 1]
