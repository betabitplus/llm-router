from __future__ import annotations

import pytest

from llm_router import Provider
from llm_router._internal.config import build_default_config
from llm_router._internal.runtime.limiter import KeyResolver

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.verifies("REQ_CREDENTIAL_RESOLUTION[revision==1]")
def test_auto_rotation_ignores_numeric_only_environment_names(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    config = build_default_config()
    for number in range(40):
        monkeypatch.delenv(f"NVIDIA_API_KEY_{number}", raising=False)
    monkeypatch.setenv("NVIDIA_API_KEY_1", "v1")
    monkeypatch.setenv("NVIDIA_API_KEY_2", "v2")
    monkeypatch.setenv("7", "unrelated")
    resolver = KeyResolver(config)

    candidate_ids = [
        item.key_id
        for item in resolver.candidates(provider=Provider.NVIDIA, key_id="auto")
    ]
    resolved = [
        resolver.resolve(provider=Provider.NVIDIA, key_id="auto") for _ in range(4)
    ]

    assert candidate_ids == [1, 2]
    assert [item.key_id for item in resolved] == [1, 2, 1, 2]
