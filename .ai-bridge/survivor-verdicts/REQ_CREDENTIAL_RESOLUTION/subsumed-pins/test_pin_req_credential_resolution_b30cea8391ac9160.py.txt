from __future__ import annotations

import pytest

from llm_router import Provider
from llm_router._internal.config import build_default_config
from llm_router._internal.runtime.limiter import KeyResolver

pytestmark = pytest.mark.verification_kind("unit")


@pytest.mark.verifies("REQ_CREDENTIAL_RESOLUTION[revision==1]")
def test_fixed_missing_optional_credential_resolves_empty_value(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    provider = Provider.QWENCHAT
    key_id = 9
    env_name = f"{provider.name}_API_KEY_{key_id}"
    monkeypatch.delenv(env_name, raising=False)

    resolved = KeyResolver(build_default_config()).resolve(
        provider=provider, key_id=key_id
    )

    assert resolved is not None
    assert resolved.key_id == key_id
    assert resolved.env_var == env_name
    assert resolved.value == ""
