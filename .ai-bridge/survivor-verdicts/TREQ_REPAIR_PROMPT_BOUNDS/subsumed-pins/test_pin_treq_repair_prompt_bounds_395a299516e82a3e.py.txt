from __future__ import annotations

import pytest
from hypothesis import given, settings, strategies as st

from llm_router._internal.capabilities.schema import (
    SchemaSpec,
    build_repair_prompt,
)

pytestmark = pytest.mark.verification_kind("unit")

_PREFIX = "\nRequired schema preview: "
_NEXT = "\nValidation error: "


def _identity(value: object) -> object:
    return value


@pytest.mark.verifies("TREQ_REPAIR_PROMPT_BOUNDS[revision==2]")
@settings(max_examples=20, deadline=None)
@given(count=st.integers(min_value=3_000, max_value=8_000))
def test_huge_schema_preview_is_capped_at_500(count: int) -> None:
    schema = {
        "type": "object",
        "properties": {
            f"field_{index}": {"type": "string", "description": "d" * 20}
            for index in range(count)
        },
    }
    spec = SchemaSpec(name="n" * count, json_schema=schema, parser=_identity)
    prompt = build_repair_prompt(
        spec=spec,
        invalid_output="o" * (count * 10),
        error_message="e" * (count * 10),
    )
    start = prompt.index(_PREFIX) + len(_PREFIX)
    segment = prompt[start : prompt.index(_NEXT, start)]
    assert len(segment) <= 500
    assert len(segment) >= 400
    assert len(prompt) <= 1_800
